# F-3.6 volatile functions recalculate every pass. NOW is a test-runner builtin.
sheet Sheet1

set A1 1
set B1 =A1+1
set C1 =NOW()

# After the first recalc C1 is 1 (mock NOW returns the pass counter).
expect C1 1
expect B1 2

# A second recalc without edits must still refresh NOW, not B1's formula identity.
# The runner performs an extra incremental pass and asserts C1 advanced.
expect_volatile C1
